<script>
(function () {
  "use strict";

  var root = document.getElementById("cmp-blog-directory");
  if (!root) return;

  var filters = root.querySelector(".cmp-directory-filters");
  var search = root.querySelector(".cmp-directory-search");
  var status = root.querySelector(".cmp-directory-status");
  var content = root.querySelector(".cmp-directory-content");
  var activeFilter = "all";
  var directoryData = null;
  var alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ".split("");
  var groupOrder = ["0-9"].concat(alphabet).concat(["other-languages"]);

  function groupLabel(group) {
    return group === "other-languages" ? "Other Languages" : group;
  }

  function normalise(value) {
    return String(value || "").toLocaleLowerCase("en-GB");
  }

  function makeFilter(value, label, enabled) {
    var button = document.createElement("button");
    button.type = "button";
    button.className = "cmp-directory-filter";
    button.dataset.filter = value;
    button.textContent = label;
    button.disabled = !enabled;
    button.setAttribute("aria-pressed", value === activeFilter ? "true" : "false");
    button.addEventListener("click", function () {
      activeFilter = value;
      Array.prototype.forEach.call(filters.querySelectorAll("button"), function (item) {
        item.setAttribute("aria-pressed", item.dataset.filter === activeFilter ? "true" : "false");
      });
      renderEntries();
    });
    return button;
  }

  function buildFilters(blogs) {
    var available = new Set(blogs.map(function (blog) { return blog.initial; }));
    filters.replaceChildren();
    filters.appendChild(makeFilter("all", "All", true));
    filters.appendChild(makeFilter("0-9", "0–9", available.has("0-9")));
    alphabet.forEach(function (letter) {
      filters.appendChild(makeFilter(letter, letter, available.has(letter)));
    });
    filters.appendChild(
      makeFilter("other-languages", "Other Languages", available.has("other-languages"))
    );
  }

  function makeBlogItem(blog) {
    var item = document.createElement("li");
    var link = document.createElement("a");
    link.href = blog.site_url;
    link.target = "_blank";
    link.rel = "noopener noreferrer";
    link.textContent = blog.name;
    item.appendChild(link);

    if (blog.directory_group === "other-languages" && blog.language) {
      var language = document.createElement("span");
      language.className = "cmp-directory-language";
      language.textContent = "(" + blog.language + ")";
      item.appendChild(language);
    }
    return item;
  }

  function renderEntries() {
    if (!directoryData) return;

    var query = normalise(search.value).trim();
    var blogs = directoryData.blogs.filter(function (blog) {
      var filterMatch = activeFilter === "all" || blog.initial === activeFilter;
      var searchMatch = !query || normalise(blog.name).indexOf(query) !== -1;
      return filterMatch && searchMatch;
    });

    content.replaceChildren();
    groupOrder.forEach(function (group) {
      var groupBlogs = blogs.filter(function (blog) { return blog.initial === group; });
      if (!groupBlogs.length) return;

      var section = document.createElement("section");
      section.className = "cmp-directory-section";
      section.id = "cmp-directory-" + group.toLocaleLowerCase();

      var heading = document.createElement("h3");
      heading.textContent = groupLabel(group);
      section.appendChild(heading);

      var list = document.createElement("ul");
      list.className = "cmp-directory-list";
      groupBlogs.forEach(function (blog) { list.appendChild(makeBlogItem(blog)); });
      section.appendChild(list);
      content.appendChild(section);
    });

    var total = directoryData.blog_count || directoryData.blogs.length;
    status.textContent = blogs.length === total && !query && activeFilter === "all"
      ? total + " blogs listed."
      : "Showing " + blogs.length + " of " + total + " blogs.";
    content.hidden = blogs.length === 0;

    if (!blogs.length) {
      status.textContent = "No blogs match the current filter.";
    }
  }

  window.__cmpRenderBlaugustBlogDirectory = function (data) {
    var blogs = data && Array.isArray(data.blogs) ? data.blogs : [];
    if (!blogs.length) {
      status.textContent = "The blog directory is currently unavailable.";
      return;
    }

    directoryData = data;
    buildFilters(blogs);
    search.addEventListener("input", renderEntries);
    renderEntries();
  };

  if (window.__blaugustBlogDirectoryData) {
    window.__cmpRenderBlaugustBlogDirectory(window.__blaugustBlogDirectoryData);
    return;
  }

  var script = document.createElement("script");
  script.src = root.getAttribute("data-source") + "?v=" + Date.now();
  script.async = true;
  script.onerror = function () {
    status.textContent = "The blog directory could not be loaded at present.";
  };
  document.head.appendChild(script);
})();
</script>
